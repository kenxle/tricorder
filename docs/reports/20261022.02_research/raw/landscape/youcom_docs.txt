Python
from youdotcom import You
from youdotcom.models import Extraction, ExtractionMode
with You() as you:
 res = you.search(
 query="best practices for scaling microservices architecture in production",
 extraction=Extraction(extraction_mode=ExtractionMode.HIGHLIGHTS),
 )
 # Each result contains a title, URL, query-relevant highlights, and more
 if res.results and res.results.web:
 for result in res.results.web:
 print(f"Title: {result.title}")
 print(f"URL: {result.url}")
 if result.contents and result.contents.highlights:
 print(f"Highlight: {result.contents.highlights[0]}\n")
